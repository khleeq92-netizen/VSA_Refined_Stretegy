// UHV Sweep Reversal — RESEARCH strategy for Deep Backtesting on OANDA:XAUUSD M5.
// It reproduces the Python research model (research/backtest_v2.py + research/uhv_indicator.py).
// Purpose: validation over many years. It is NOT a validated live-trading system.
//
// BUY  : red UHV candle at exhaustion (your indicator's Part-4 rules) -> later candle sweeps the UHV low
//        (while price is still inside the UHV range) -> a candle AFTER the sweep is bullish and closes above
//        the UHV high (and above the CISD candle when CISD is on) -> enter next bar open.
//        SL = lowest low since the UHV candle - buffer. 50% at TP1 (1R), rest at TP2 (2R).
// SELL : exact mirror (green UHV at the top, sweep of the UHV high, bearish close below the UHV low).
// All signals are computed on CONFIRMED bars; higher-timeframe zones use completed HTF candles only.
//
// For results to match the research, run BUY-only and SELL-only as separate backtests
// (the Python engine tests each side independently).
//@version=6
strategy("UHV Sweep Reversal [research]", overlay = true, initial_capital = 100000,
     default_qty_type = strategy.fixed, default_qty_value = 1, pyramiding = 0,
     commission_type = strategy.commission.cash_per_contract, commission_value = 0.20,
     slippage = 0, process_orders_on_close = false, calc_on_every_tick = false,
     max_labels_count = 500, max_lines_count = 500)

// ───────── General
gG = "General"
dirMode   = input.string("Buy only", "Direction", options = ["Buy only", "Sell only", "Both"], group = gG,
     tooltip = "Run Buy only and Sell only as separate backtests to match the research.")
startT    = input.time(timestamp("2021-01-01T00:00:00"), "Backtest start", group = gG)
endT      = input.time(timestamp("2030-01-01T00:00:00"), "Backtest end", group = gG)
riskUSD   = input.float(100, "Risk per trade ($) — profit / this = R", minval = 1, group = gG)
sessMode  = input.string("US session", "Trade in", options = ["US session", "All sessions"], group = gG)
usSess    = input.session("0800-1700", "US session (New York time)", group = gG)

// ───────── UHV detector (from your indicator guide, Part 4)
gU = "UHV detector"
vmBuy     = input.float(2.0, "BUY UHV multiplier (x avg volume)", step = 0.1, group = gU)
vmSell    = input.float(2.0, "SELL UHV multiplier (x avg volume)", step = 0.1, group = gU)
nAvg      = input.int(40, "Lookback for volume / size averages", group = gU)
bodyK     = input.float(1.2, "Body >= x average body", step = 0.1, group = gU)
rangeK    = input.float(1.2, "Range >= x average range", step = 0.1, group = gU)
posN      = input.int(50, "Range lookback for location test", group = gU)
posMax    = input.float(0.65, "BUY: max position in range (0=low,1=high); SELL uses 1 - this", step = 0.05, group = gU)
runN      = input.int(6, "Run: look back this many candles", group = gU)
runMin    = input.int(4, "Run: at least this many against-colour candles", group = gU)
runVol    = input.float(0.8, "Run: avg volume >= x average", step = 0.1, group = gU)
useOv     = input.bool(true, "Exceptional volume can override the run rule", group = gU)
excK      = input.float(4.0, "Exceptional = x average volume", step = 0.5, group = gU)
excN      = input.int(400, "...or highest volume of this many candles", group = gU)
ovReds    = input.int(2, "Against-colour candles still required on override", group = gU)
bgN       = input.int(12, "Background lookback", group = gU)
emaN      = input.int(50, "Background EMA length", group = gU)
bgShare   = input.float(0.55, "Background colour share", step = 0.05, group = gU)
bgNeed    = input.int(2, "Background tests that must pass (of 3)", minval = 1, maxval = 3, group = gU)

// ───────── Setup / trigger
gS = "Setup and trigger"
sweepMax  = input.int(288, "Max bars from UHV to sweep (setup dies earlier if price closes beyond the other UHV extreme)", group = gS)
trigMax   = input.int(12, "Max bars from sweep to trigger", group = gS)
useCisd   = input.bool(false, "Require close beyond the CISD candle too", group = gS)
cisdRef   = input.string("High/Low", "CISD candle level", options = ["High/Low", "Open"], group = gS)
minBody   = input.float(0.0, "Trigger candle min body % of range (0 = off)", step = 0.1, group = gS)
slBuf     = input.float(0.10, "SL buffer beyond the extreme ($)", step = 0.05, group = gS)
minRisk   = input.float(0.20, "Skip trades with risk below ($)", step = 0.05, group = gS)

// ───────── Higher-timeframe zone filter
gZ = "HTF zone filter (UHV candle must overlap an active zone)"
useZone   = input.bool(false, "Require HTF zone", group = gZ, tooltip = "Off by default: in the Aug-Nov 2026 research sample it reduced trades without improving results. Test both.")
zM15F     = input.bool(true, "M15 FVG", group = gZ, inline = "z1")
zH1F      = input.bool(true, "H1 FVG", group = gZ, inline = "z1")
zH4F      = input.bool(true, "H4 FVG", group = gZ, inline = "z1")
zM15O     = input.bool(true, "M15 demand/supply", group = gZ, inline = "z2")
zH1O      = input.bool(true, "H1 demand/supply", group = gZ, inline = "z2")
zH4O      = input.bool(true, "H4 demand/supply", group = gZ, inline = "z2")
zoneDays  = input.int(5, "Zone max age (days)", group = gZ)

// ───────── Exits
gX = "Exits"
tp1R      = input.float(1.0, "TP1 (R)", step = 0.25, group = gX)
tp2R      = input.float(2.0, "Final TP (R)", step = 0.25, group = gX)
tp1Pct    = input.float(50, "% closed at TP1 (0 = single target at final TP)", minval = 0, maxval = 100, group = gX)
maxHold   = input.int(96, "Time exit after N bars", group = gX)

// ═════════ UHV detection (confirmed bar)
vAvg   = ta.sma(volume, nAvg)
bodyV  = math.abs(close - open)
bigOK  = bodyV >= bodyK * ta.sma(bodyV, nAvg) and (high - low) >= rangeK * ta.sma(high - low, nAvg)
hhP    = ta.highest(high, posN)
llP    = ta.lowest(low, posN)
posR   = hhP > llP ? (close - llP) / (hhP - llP) : 0.5
excV   = volume >= excK * vAvg or volume >= ta.highest(volume, excN)
emaV   = ta.ema(close, emaN)

f_uhv(bool isBuy, float vm) =>
    bool  against = isBuy ? close < open : close > open
    float ag1  = (isBuy ? close[1] < open[1] : close[1] > open[1]) ? 1.0 : 0.0
    float cnt  = math.sum(ag1, runN)
    float runV = cnt > 0 ? math.sum(ag1 * volume[1], runN) / cnt : 0.0
    bool  normalOK = cnt >= runMin and runV >= runVol * vAvg
    float t1 = (isBuy ? close < emaV : close > emaV) ? 1 : 0
    float t2 = (isBuy ? close < close[bgN] : close > close[bgN]) ? 1 : 0
    float t3 = math.sum(against ? 1.0 : 0.0, bgN) / bgN >= bgShare ? 1 : 0
    bool  ovOK = useOv and excV and (t1 + t2 + t3) >= bgNeed and cnt >= ovReds
    bool  locOK = isBuy ? posR <= posMax : posR >= 1 - posMax
    against and volume > vm * vAvg and bigOK and locOK and (normalOK or ovOK)

uhvBuy  = f_uhv(true, vmBuy)
uhvSell = f_uhv(false, vmSell)

// ═════════ HTF zones from COMPLETED higher-timeframe candles (lookahead_on + [1] offsets = no repaint)
f_htf() =>
    float atrP = ta.sma(ta.tr(true), 14)[2]
    bool  bullF = low[1] > high[3]
    bool  bearF = high[1] < low[3]
    float dB = na
    float dT = na
    float sB = na
    float sT = na
    if high[1] - low[1] >= 1.5 * atrP
        if close[1] > open[1]
            for j = 2 to 6
                if close[j] < open[j]
                    if close[1] > high[j]
                        dB := low[j]
                        dT := high[j]
                    break
        if close[1] < open[1]
            for j = 2 to 6
                if close[j] > open[j]
                    if close[1] < low[j]
                        sB := low[j]
                        sT := high[j]
                    break
    [bullF, high[3], low[1], bearF, high[1], low[3], dB, dT, sB, sT, time[1]]

[aBF, aBb, aBt, aSF, aSb, aSt, aDB, aDT, aSB, aST, aT] = request.security(syminfo.tickerid, "15",  f_htf(), lookahead = barmerge.lookahead_on)
[bBF, bBb, bBt, bSF, bSb, bSt, bDB, bDT, bSB, bST, bT] = request.security(syminfo.tickerid, "60",  f_htf(), lookahead = barmerge.lookahead_on)
[cBF, cBb, cBt, cSF, cSb, cSt, cDB, cDT, cSB, cST, cT] = request.security(syminfo.tickerid, "240", f_htf(), lookahead = barmerge.lookahead_on)

var zBot  = array.new_float()
var zTop  = array.new_float()
var zSide = array.new_int()
var zExp  = array.new_int()
expMs = zoneDays * 86400000

f_add(float b, float t, int sd) =>
    if not na(b) and not na(t) and t > b
        array.push(zBot, b)
        array.push(zTop, t)
        array.push(zSide, sd)
        array.push(zExp, time + expMs)

if useZone
    if ta.change(aT) != 0
        if zM15F and aBF
            f_add(aBb, aBt, 1)
        if zM15F and aSF
            f_add(aSb, aSt, -1)
        if zM15O
            f_add(aDB, aDT, 1)
            f_add(aSB, aST, -1)
    if ta.change(bT) != 0
        if zH1F and bBF
            f_add(bBb, bBt, 1)
        if zH1F and bSF
            f_add(bSb, bSt, -1)
        if zH1O
            f_add(bDB, bDT, 1)
            f_add(bSB, bST, -1)
    if ta.change(cT) != 0
        if zH4F and cBF
            f_add(cBb, cBt, 1)
        if zH4F and cSF
            f_add(cSb, cSt, -1)
        if zH4O
            f_add(cDB, cDT, 1)
            f_add(cSB, cST, -1)

bool hitBull = false
bool hitBear = false
if useZone and array.size(zBot) > 0
    for i = array.size(zBot) - 1 to 0
        float b = array.get(zBot, i)
        float t = array.get(zTop, i)
        int   sd = array.get(zSide, i)
        bool  dead = time > array.get(zExp, i)
        if not dead
            bool ov = low <= t and high >= b
            if sd == 1
                hitBull := hitBull or ov
                dead := close < b
            else
                hitBear := hitBear or ov
                dead := close > t
        if dead
            array.remove(zBot, i)
            array.remove(zTop, i)
            array.remove(zSide, i)
            array.remove(zExp, i)

// ═════════ Filters
inDate  = time >= startT and time < endT
inSess  = sessMode == "All sessions" or not na(time(timeframe.period, usSess, "America/New_York"))
flat    = strategy.position_size == 0
allowB  = dirMode != "Sell only"
allowS  = dirMode != "Buy only"
bodyPct = high > low ? math.abs(close - open) / (high - low) : 0.0

// CISD candle: first candle of the against-colour run ending at the UHV candle; if it is an inside
// candle, step back to its mother candle. Level = its high (BUY) / low (SELL), or its open.
f_cisd(bool isBuy) =>
    int j = 0
    for k = 1 to 30
        bool ag = isBuy ? close[k] < open[k] : close[k] > open[k]
        if not ag
            break
        j := k
    for k = 1 to 10
        if high[j] <= high[j + 1] and low[j] >= low[j + 1]
            j := j + 1
        else
            break
    cisdRef == "Open" ? open[j] : (isBuy ? high[j] : low[j])

// ═════════ BUY setup state machine
var bool  bAct = false
var bool  bSw = false
var int   bU = na
var int   bSwBar = na
var float bH = na
var float bL = na
var float bLow = na
var float bLvl = na
bool goLong = false
float longSL = na

if bAct and bar_index > bU
    bLow := math.min(bLow, low)
    if not bSw
        if low < bL
            bSw := true
            bSwBar := bar_index
        else if close > bH or bar_index - bU >= sweepMax
            bAct := false
    else if bar_index > bSwBar
        if bar_index - bSwBar > trigMax
            bAct := false
        else if close > open and close > bLvl and bodyPct >= minBody
            if inSess and inDate and flat and allowB and close - (bLow - slBuf) > minRisk
                goLong := true
                longSL := bLow - slBuf
            bAct := false

if not bAct and uhvBuy and flat and allowB and (not useZone or hitBull)
    bAct := true
    bSw := false
    bU := bar_index
    bH := high
    bL := low
    bLow := low
    bLvl := useCisd ? math.max(high, f_cisd(true)) : high

// ═════════ SELL setup state machine (mirror)
var bool  sAct = false
var bool  sSw = false
var int   sU = na
var int   sSwBar = na
var float sH = na
var float sL = na
var float sHigh = na
var float sLvl = na
bool goShort = false
float shortSL = na

if sAct and bar_index > sU
    sHigh := math.max(sHigh, high)
    if not sSw
        if high > sH
            sSw := true
            sSwBar := bar_index
        else if close < sL or bar_index - sU >= sweepMax
            sAct := false
    else if bar_index > sSwBar
        if bar_index - sSwBar > trigMax
            sAct := false
        else if close < open and close < sLvl and bodyPct >= minBody
            if inSess and inDate and flat and allowS and (sHigh + slBuf) - close > minRisk
                goShort := true
                shortSL := sHigh + slBuf
            sAct := false

if not sAct and uhvSell and flat and allowS and (not useZone or hitBear)
    sAct := true
    sSw := false
    sU := bar_index
    sH := high
    sL := low
    sHigh := high
    sLvl := useCisd ? math.min(low, f_cisd(false)) : low

// ═════════ Orders (placed at the signal bar's close, filled at the next bar's open)
var float slP = na
var float tp1P = na
var float tp2P = na
var int   sigBar = na

if goLong and goShort
    goShort := false   // "Both" mode: one position at a time, BUY wins a tie

if goLong
    float risk = close - longSL
    slP := longSL
    tp1P := close + tp1R * risk
    tp2P := close + tp2R * risk
    sigBar := bar_index
    strategy.entry("L", strategy.long, qty = riskUSD / risk)
    if tp1Pct > 0
        strategy.exit("L-TP1", "L", qty_percent = tp1Pct, limit = tp1P, stop = slP)
    strategy.exit("L-TP2", "L", limit = tp2P, stop = slP)
    alert("UHV BUY: entry ~" + str.tostring(close, format.mintick) + " SL " + str.tostring(slP, format.mintick) +
         " TP1 " + str.tostring(tp1P, format.mintick) + " TP2 " + str.tostring(tp2P, format.mintick), alert.freq_once_per_bar_close)

if goShort
    float risk = shortSL - close
    slP := shortSL
    tp1P := close - tp1R * risk
    tp2P := close - tp2R * risk
    sigBar := bar_index
    strategy.entry("S", strategy.short, qty = riskUSD / risk)
    if tp1Pct > 0
        strategy.exit("S-TP1", "S", qty_percent = tp1Pct, limit = tp1P, stop = slP)
    strategy.exit("S-TP2", "S", limit = tp2P, stop = slP)
    alert("UHV SELL: entry ~" + str.tostring(close, format.mintick) + " SL " + str.tostring(slP, format.mintick) +
         " TP1 " + str.tostring(tp1P, format.mintick) + " TP2 " + str.tostring(tp2P, format.mintick), alert.freq_once_per_bar_close)

if not flat and not na(sigBar) and bar_index - sigBar >= maxHold
    strategy.close_all(comment = "time")

// ═════════ Visuals
inPos = not flat or goLong or goShort
plot(inPos ? slP : na, "SL", color.red, 1, plot.style_linebr)
plot(inPos ? tp1P : na, "TP1", color.new(color.green, 40), 1, plot.style_linebr)
plot(inPos ? tp2P : na, "TP2", color.green, 2, plot.style_linebr)
plotshape(uhvBuy and allowB, "UHV buy candle", shape.triangleup, location.belowbar, color.new(color.teal, 0), size = size.tiny)
plotshape(uhvSell and allowS, "UHV sell candle", shape.triangledown, location.abovebar, color.new(color.maroon, 0), size = size.tiny)
plotshape(bAct and bSw and bar_index == bSwBar, "Buy sweep", shape.xcross, location.belowbar, color.teal, size = size.tiny)
plotshape(sAct and sSw and bar_index == sSwBar, "Sell sweep", shape.xcross, location.abovebar, color.maroon, size = size.tiny)
